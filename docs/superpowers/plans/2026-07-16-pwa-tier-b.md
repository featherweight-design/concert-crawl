# PWA Tier B — Installable + Offline Browsing — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn `apps/web` into a genuinely installable, offline-capable PWA that caches public data (events, venues) offline, with a functional install affordance and offline indicator.

**Architecture:** Add a Serwist service worker via `@serwist/next`, wired through `next.config.ts`. Complete the manifest + icon/favicon set + iOS meta. Runtime-cache public API GETs stale-while-revalidate; keep all authenticated routes network-only. Ship a minimal (design-polish-deferred) offline banner + install prompt. Verify with Playwright e2e + a Lighthouse CI check.

**Tech Stack:** Next.js 15.5.20 (App Router), React 19, TypeScript strict, Tailwind v4, Serwist (`@serwist/next` + `serwist`), Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-07-16-pwa-tier-b-design.md`

## Global Constraints

- Serwist only — never `next-pwa` (unmaintained).
- Cache **public** data only (events, venues). All authenticated API routes are **network-only, never cached** — enforced by an explicit allowlist that defaults to network-only.
- TypeScript strict; **no `as` casts** — use type guards / narrowing / declaration merging.
- All customer-facing strings live in a `definitions/copy.ts` file.
- Components: arrow functions, no `handle*` names, logic in `lib/`, semantic HTML, a11y (keyboard/ARIA/44px touch), no native `title=` tooltips.
- Mobile/desktop parity in the same PR.
- Run `pnpm --filter @concert-crawl/web typecheck && pnpm --filter @concert-crawl/web lint && pnpm --filter @concert-crawl/web test` before every push.
- Serwist disabled in development (`disable: process.env.NODE_ENV === 'development'`).
- This effort ships as 4 sequential PRs (phases). Each PR passes CI on its own.

- [ ] **Step 1: Extend the component signature**

Maskable icons must keep their art inside a safe zone (outer ~10% may be cropped to a circle/squircle). Render the existing art group scaled to 0.8 and centered on a full-bleed canvas fill.

```tsx
// apps/web/src/app/api/pwa/icon-svg.tsx — replace the interface + component
interface PwaIconSvgProps {
  size: number
  maskable?: boolean
}

export function PwaIconSvg({ size, maskable = false }: PwaIconSvgProps) {
  // Maskable: full-bleed canvas + art inset into the 80% safe zone.
  const artTransform = maskable ? 'translate(4.8 4.8) scale(0.8)' : undefined
  return (
    <svg
      width={size}
      height={size}
      viewBox="0 0 48 48"
      xmlns="http://www.w3.org/2000/svg"
    >
      <rect width="48" height="48" fill={PWA_ICON_PALETTE.canvas} />
      <g transform={artTransform}>
        {!maskable && (
          <rect width="48" height="48" rx="10" fill={PWA_ICON_PALETTE.background} />
        )}
        {maskable && (
          <rect width="48" height="48" fill={PWA_ICON_PALETTE.background} />
        )}
        {/* Signal ring */}
        <circle cx="34" cy="14" r="9" fill={PWA_ICON_PALETTE.signalOuter} opacity="0.22" />
        <circle cx="34" cy="14" r="5" fill={PWA_ICON_PALETTE.signalOuter} opacity="0.45" />
        <circle cx="34" cy="14" r="3" fill={PWA_ICON_PALETTE.signalInner} />
        {/* Dot trail */}
        <circle cx="14" cy="38" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.45" />
        <circle cx="18" cy="34" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.52" />
        <circle cx="22" cy="30" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.59" />
        <circle cx="26" cy="26" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.66" />
        <circle cx="30" cy="22" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.73" />
        <circle cx="34" cy="18" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.80" />
        <circle cx="14" cy="32" r="1.6" fill={PWA_ICON_PALETTE.dot} opacity="0.87" />

  )
}
```

```tsx
'use client'

import { useEffect } from 'react'

export const ServiceWorkerRegistration = () => {
  useEffect(() => {
    if (!('serviceWorker' in navigator)) return
    if (process.env.NODE_ENV !== 'production') return

    const register = async () => {
      try {
        await navigator.serviceWorker.register('/sw.js', { scope: '/' })
      } catch (err) {
        // Registration failures must never break the app shell.
        console.error('Service worker registration failed', err)
      }
    }

    void register()
  }, [])

  return null
}
```

```ts
// apps/web/src/lib/pwa/route-cache.ts

// Explicit allowlist of PUBLIC, cacheable API path prefixes. Anything not
// matched here is network-only — this is the privacy safeguard, so the
// default must stay deny.
const PUBLIC_API_PREFIXES = ['/v1/events', '/v1/venues'] as const

export const isPublicCacheableApi = (url: URL): boolean =>
  PUBLIC_API_PREFIXES.some(
    (prefix) => url.pathname === prefix || url.pathname.startsWith(`${prefix}/`),
  )
```

```ts
'use client'

import { useEffect, useState } from 'react'

export const useOnlineStatus = (): boolean => {
  const [online, setOnline] = useState(true)

  useEffect(() => {
    setOnline(navigator.onLine)
    const goOnline = () => setOnline(true)
    const goOffline = () => setOnline(false)
    window.addEventListener('online', goOnline)
    window.addEventListener('offline', goOffline)
    return () => {
      window.removeEventListener('online', goOnline)
      window.removeEventListener('offline', goOffline)
    }
  }, [])

  return online
}
```

```ts
// apps/web/src/lib/pwa/beforeinstallprompt.ts

// The beforeinstallprompt event is not in the DOM lib. Declare it via
// interface merging rather than casting.
export interface BeforeInstallPromptEvent extends Event {
  readonly platforms: string[]
  readonly userChoice: Promise<{ outcome: 'accepted' | 'dismissed' }>
  prompt: () => Promise<void>
}

declare global {
  interface WindowEventMap {
    beforeinstallprompt: BeforeInstallPromptEvent
  }
}

export const isIosSafari = (): boolean => {
  if (typeof navigator === 'undefined') return false
  const ua = navigator.userAgent
  const isIos = /iPad|iPhone|iPod/.test(ua)
  const isSafari = /Safari/.test(ua) && !/CriOS|FxiOS|EdgiOS/.test(ua)
  return isIos && isSafari
}

export const isStandalone = (): boolean => {
  if (typeof window === 'undefined') return false
  // iOS Safari exposes standalone via navigator.standalone (no display-mode support
  // on older versions); check both so the install hint hides once installed.
  // Annotate (not cast) to reach the non-standard prop without an `as`.
  const nav: Navigator & { standalone?: boolean } = window.navigator
  return window.matchMedia('(display-mode: standalone)').matches || nav.standalone === true
}
```

- [ ] **Step 2: Create the install prompt component**

```tsx
// apps/web/src/components/pwa/InstallPrompt.tsx
'use client'

import { useEffect, useState } from 'react'
import {
  isIosSafari,
  isStandalone,
  type BeforeInstallPromptEvent,
} from '@/lib/pwa/beforeinstallprompt'
import { PWA_COPY } from './definitions/copy'

const DISMISS_KEY = 'cc-install-dismissed'

export const InstallPrompt = () => {
  const [deferred, setDeferred] = useState<BeforeInstallPromptEvent | null>(null)
  const [showIosHint, setShowIosHint] = useState(false)
  const [dismissed, setDismissed] = useState(true)

  useEffect(() => {
    if (isStandalone()) return
    setDismissed(localStorage.getItem(DISMISS_KEY) === '1')

    const onPrompt = (event: BeforeInstallPromptEvent) => {
      event.preventDefault()
      setDeferred(event)
    }
    window.addEventListener('beforeinstallprompt', onPrompt)
    if (isIosSafari()) setShowIosHint(true)

    return () => window.removeEventListener('beforeinstallprompt', onPrompt)
  }, [])

  const dismiss = () => {
    localStorage.setItem(DISMISS_KEY, '1')
    setDismissed(true)
  }

  const install = async () => {
    if (!deferred) return
    await deferred.prompt()
    await deferred.userChoice
    setDeferred(null)
  }

  if (dismissed) return null
  if (!deferred && !showIosHint) return null

  return (
    <div
      role="region"
      aria-label={PWA_COPY.install}
      className="fixed inset-x-4 bottom-4 z-50 flex items-center justify-between gap-3 rounded-lg bg-surface p-3 shadow-lg"
    >
      <span className="text-sm text-fg">
        {deferred ? PWA_COPY.install : PWA_COPY.installIosHint}
      </span>
      <div className="flex items-center gap-2">
        {deferred && (
          <button
            type="button"
            onClick={install}
            className="inline-flex min-h-[44px] items-center rounded-md bg-accent px-4 py-2 text-sm text-accent-fg"
          >
            {PWA_COPY.install}
          </button>
        )}
        <button
          type="button"
          onClick={dismiss}
          aria-label={PWA_COPY.dismiss}
          className="inline-flex min-h-[44px] min-w-[44px] items-center justify-center rounded-md px-3 py-2 text-sm text-fg-muted"
        >
          {PWA_COPY.dismiss}
        </button>
      </div>
    </div>
  )
}
```
