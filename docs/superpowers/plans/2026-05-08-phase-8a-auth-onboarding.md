```ts
// apps/api/src/lib/rate-limit.ts
interface RateLimitEntry {
  count: number
  resetAt: number
}

const store = new Map<string, RateLimitEntry>()

export function checkRateLimit(
  key: string,
  maxRequests: number,
  windowMs: number,
): { allowed: boolean; retryAfterMs: number } {
  const now = Date.now()
  const entry = store.get(key)

  if (!entry || now >= entry.resetAt) {
    store.set(key, { count: 1, resetAt: now + windowMs })
    return { allowed: true, retryAfterMs: 0 }
  }

  if (entry.count >= maxRequests) {
    return { allowed: false, retryAfterMs: entry.resetAt - now }
  }

  entry.count += 1
  return { allowed: true, retryAfterMs: 0 }
}
```

```ts
  email: z.string().email(),

  '/',

    // Always return 200 — no email enumeration

  },
```

```ts
// apps/api/src/routes/auth/validate-code.ts
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { z } from 'zod'
import { db, inviteCodes } from '@concert-crawl/db'
import { eq } from 'drizzle-orm'
import { checkRateLimit } from '../../lib/rate-limit.js'

const codeSchema = z.object({
  code: z.string().regex(/^BETA-[A-Z0-9]{4}-[A-Z0-9]{4}$/, 'Invalid code format'),
})

export const validateCodeRouter = new Hono()

validateCodeRouter.post(
  '/',
  zValidator('json', codeSchema),
  async (ctx) => {
    const ip = ctx.req.header('x-forwarded-for')?.split(',')[0]?.trim() ?? 'unknown'
    const limit = checkRateLimit(`validate-code:${ip}`, 10, 60 * 60 * 1000)
    if (!limit.allowed) {
      return ctx.json({ success: false, error: 'ratelimit' }, 429)
    }

    const { code } = ctx.req.valid('json')

    const [invite] = await db
      .select()
      .from(inviteCodes)
      .where(eq(inviteCodes.code, code))

    if (!invite) {
      return ctx.json({ success: false, error: 'notfound' }, 404)
    }
    if (invite.revokedAt) {
      return ctx.json({ success: false, error: 'notfound' }, 404)
    }
    if (invite.expiresAt && invite.expiresAt < new Date()) {
      return ctx.json({ success: false, error: 'expired' }, 410)
    }
    if (invite.usedBy) {
      return ctx.json({ success: false, error: 'used' }, 409)
    }

    return ctx.json({ ok: true })
  },
)
```

```ts
// apps/api/src/routes/auth/redeem-code.ts
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { z } from 'zod'
import { db, inviteCodes } from '@concert-crawl/db'
import { eq } from 'drizzle-orm'
import { clerkAuth, requireAuth, getAuthUser } from '../../middleware/auth.js'

const redeemSchema = z.object({
  code: z.string().regex(/^BETA-[A-Z0-9]{4}-[A-Z0-9]{4}$/),
})

export const redeemCodeRouter = new Hono()
  .use(clerkAuth)
  .use(requireAuth)

redeemCodeRouter.post('/', zValidator('json', redeemSchema), async (ctx) => {
  const { userId } = getAuthUser(ctx)
  const { code } = ctx.req.valid('json')

  const [invite] = await db
    .select()
    .from(inviteCodes)
    .where(eq(inviteCodes.code, code))

  if (!invite) {
    return ctx.json({ success: false, error: 'notfound' }, 404)
  }

  // Idempotent: already redeemed by this user
  if (invite.usedBy === userId) {
    return ctx.json({ ok: true })
  }

  // Already redeemed by someone else
  if (invite.usedBy) {
    return ctx.json({ success: false, error: 'used' }, 409)
  }

  await db
    .update(inviteCodes)
    .set({ usedBy: userId, usedAt: new Date() })
    .where(eq(inviteCodes.id, invite.id))

  return ctx.json({ ok: true })
})
```

```ts
// apps/web/src/lib/invite-cookie.ts
// Server-only — uses Node.js crypto. Do not import in middleware.ts (Edge Runtime).
import { createHmac, timingSafeEqual } from 'crypto'

export const INVITE_COOKIE_NAME = 'invite_verified'
export const INVITE_COOKIE_MAX_AGE = 3600 // 1 hour in seconds

export function signInviteCookie(code: string, secret: string): string {
  const iat = Math.floor(Date.now() / 1000)
  const payload = JSON.stringify({ code, iat })
  const payloadB64 = Buffer.from(payload).toString('base64url')
  const sig = createHmac('sha256', secret).update(payloadB64).digest('base64url')
  return `${payloadB64}.${sig}`
}

export function verifyInviteCookie(
  value: string,
  secret: string,
): { code: string } | null {
  try {
    const dotIdx = value.lastIndexOf('.')
    if (dotIdx === -1) return null

    const payloadB64 = value.slice(0, dotIdx)
    const sigB64 = value.slice(dotIdx + 1)

    const expectedSig = createHmac('sha256', secret)
      .update(payloadB64)
      .digest('base64url')
    const sigBuf = Buffer.from(sigB64, 'base64url')
    const expectedBuf = Buffer.from(expectedSig, 'base64url')

    if (sigBuf.length !== expectedBuf.length) return null
    if (!timingSafeEqual(sigBuf, expectedBuf)) return null

    const payloadStr = Buffer.from(payloadB64, 'base64url').toString()
    const payload = JSON.parse(payloadStr) as { code: string; iat: number }

    const now = Math.floor(Date.now() / 1000)
    if (now - payload.iat > INVITE_COOKIE_MAX_AGE) return null

    return { code: payload.code }
  } catch {
    return null
  }
}
```

```tsx
function smartLineBreak(name: string, maxLines = 3): string[] {
  const words = String(name || '').trim().split(/\s+/).filter(Boolean)
  if (words.length === 0) return []
  if (words.length === 1) return [words[0]]

  const connectors = new Set(['THE', 'AND', 'OF', 'A', 'AN', '&', 'FT', 'FT.', 'FEAT', 'FEAT.'])

  if (words.length === 2) {
    const joined = words.join(' ')
    const longest = Math.max(words[0].length, words[1].length)
    if (joined.length <= 9 && longest <= 6) return [joined]
    return [words[0], words[1]]
  }

  const groups: string[] = []
  let idx = 0
  while (idx < words.length) {
    const ww = words[idx].toUpperCase()
    if (connectors.has(ww) && idx < words.length - 1) {
      groups.push(`${words[idx]} ${words[idx + 1]}`)
      idx += 2
    } else {
      groups.push(words[idx])
      idx += 1
    }
  }

  if (groups.length <= maxLines) return groups

  while (groups.length > maxLines) {
    let bestI = 0
    let bestLen = Infinity
    for (let ii = 0; ii < groups.length - 1; ii++) {
      const len = groups[ii].length + groups[ii + 1].length + 1
      if (len < bestLen) { bestLen = len; bestI = ii }
    }
    groups.splice(bestI, 2, `${groups[bestI]} ${groups[bestI + 1]}`)
  }
  return groups
}

    return { pretty: `${months[dd.getMonth()]} ${dd.getDate()}`, dow: days[dd.getDay()] }
```

```tsx
'use client'

import { useState, useCallback, useEffect } from 'react'
import { GigPoster, pickPalette } from './GigPoster'
import type { Show } from './GigPoster'

interface GigPosterStackProps {
  shows: Show[]
  width?: number
  height?: number
  intervalMs?: number
}

export function GigPosterStack({
  shows,
  width = 320,
  height = 460,
  intervalMs = 5500,
}: GigPosterStackProps) {
  const [idx, setIdx] = useState(0)
  const [phase, setPhase] = useState<'rest' | 'advancing'>('rest')

  const advance = useCallback(() => {
    if (phase !== 'rest') return
    setPhase('advancing')
    setTimeout(() => {
      setIdx(ii => (ii + 1) % shows.length)
      setPhase('rest')
    }, 900)
  }, [phase, shows.length])

  useEffect(() => {
    if (shows.length < 2) return
    const timer = setTimeout(advance, intervalMs)
    return () => clearTimeout(timer)
  }, [advance, idx, intervalMs, shows.length])

  if (shows.length === 0) return null

  const order = [
    shows[idx % shows.length],
    shows[(idx + 1) % shows.length],
    shows[(idx + 2) % shows.length],
  ]
  const palettes = order.map(ss => pickPalette(ss.artist))

  const layers = [
    { x: 0,   y: 0,  rr: -1.5, scale: 1     },
    { x: 14,  y: 10, rr: 3.5,  scale: 0.985 },
    { x: -10, y: 20, rr: -5,   scale: 0.97  },
  ]

  const padX = Math.max(28, Math.round(width * 0.10))
  const padY = Math.max(36, Math.round(height * 0.10))

  return (
    <div
      onClick={advance}
      role="button"
      aria-label="Next poster"
      style={{
        width: width + padX * 2,
        height: height + padY * 2,
        position: 'relative',
        perspective: 1400,
        cursor: 'pointer',
      }}
    >
      {layers.map((layer, ii) => {
        const isTop = ii === 0
        const isMid = ii === 1

        let transform: string
        let opacity: number
        if (phase === 'advancing') {
          if (isTop) {
            transform = `translate(${layer.x - 24}px, ${layer.y - 110}px) rotate(${layer.rr - 12}deg) scale(${layer.scale * 1.04})`
            opacity = 0
          } else if (isMid) {
            transform = `translate(${layers[0].x}px, ${layers[0].y}px) rotate(${layers[0].rr}deg) scale(${layers[0].scale})`
            opacity = 1
          } else {
            transform = `translate(${layers[1].x}px, ${layers[1].y}px) rotate(${layers[1].rr}deg) scale(${layers[1].scale})`
            opacity = 1
          }
        } else {
          transform = `translate(${layer.x}px, ${layer.y}px) rotate(${layer.rr}deg) scale(${layer.scale})`
          opacity = 1
        }

        return (
          <div key={`${idx}-${ii}`} style={{
            position: 'absolute', left: padX, top: padY,
            zIndex: 10 - ii, transform,
            transformOrigin: '30% 100%',
            transition: phase === 'advancing'

              : 'transform 0.4s ease-out, opacity 0.4s ease-out',

          }}>
            <GigPoster show={order[ii]} width={width} height={height} palette={palettes[ii]} />
          </div>
        )
      })}
    </div>
  )
}
```

```ts
// apps/web/src/app/auth/code/actions.ts
'use server'

import { cookies } from 'next/headers'
import { redirect } from 'next/navigation'
import { signInviteCookie, INVITE_COOKIE_NAME, INVITE_COOKIE_MAX_AGE } from '@/lib/invite-cookie'

type ValidationResult =
  | { ok: true }
  | { ok: false; error: 'invalid_format' | 'notfound' | 'used' | 'expired' | 'ratelimit' | 'server_error' }

export async function validateAndSetInviteCode(code: string): Promise<ValidationResult> {
  const codePattern = /^BETA-[A-Z0-9]{4}-[A-Z0-9]{4}$/
  if (!codePattern.test(code.toUpperCase())) {
    return { ok: false, error: 'invalid_format' }
  }

  const normalizedCode = code.toUpperCase()

  try {
    const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'
    const res = await fetch(`${apiUrl}/v1/auth/validate-code`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ code: normalizedCode }),
    })

    if (res.status === 404) return { ok: false, error: 'notfound' }
    if (res.status === 409) return { ok: false, error: 'used' }
    if (res.status === 410) return { ok: false, error: 'expired' }
    if (res.status === 429) return { ok: false, error: 'ratelimit' }
    if (!res.ok) return { ok: false, error: 'server_error' }
  } catch {
    return { ok: false, error: 'server_error' }
  }

  const secret = process.env.INVITE_COOKIE_SECRET
  if (!secret) throw new Error('INVITE_COOKIE_SECRET is not set')

  const cookieValue = signInviteCookie(normalizedCode, secret)
  const cookieStore = await cookies()
  cookieStore.set(INVITE_COOKIE_NAME, cookieValue, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: INVITE_COOKIE_MAX_AGE,
    path: '/',
  })

  redirect('/onboarding')
}
```

```tsx
type StepKey = typeof FLOW[number]

function StepContent({ stepKey }: { stepKey: StepKey }) {
  if (stepKey === 'welcome') return <FounderNote />
  if (stepKey === 'how') return <HowItWorks />
  if (stepKey === 'today') return <WhatItDoesToday />
  if (stepKey === 'notyet') return <WhatItDoesntYet />
  if (stepKey === 'done') return <YoureIn />
  return null

export function OnboardingShell() {

  const router = useRouter()
  const total = FLOW.length
  const isFirst = index === 0
  const isLast = index === total - 1

    if (isLast) {
      router.push('/sign-up')

    }
    setIndex(ii => ii + 1)
```

```tsx
// apps/web/src/app/auth/complete/page.tsx
import { cookies } from 'next/headers'
import { redirect } from 'next/navigation'
import { auth } from '@clerk/nextjs/server'
import { verifyInviteCookie, INVITE_COOKIE_NAME } from '@/lib/invite-cookie'

export default async function AuthCompletePage() {
  // Must be authenticated (Clerk redirects here after sign-up)
  const { userId, getToken } = await auth()
  if (!userId) redirect('/sign-in')

  const secret = process.env.INVITE_COOKIE_SECRET
  if (!secret) {
    // Missing secret — skip redemption, go to events
    redirect('/events')
  }

  const cookieStore = await cookies()
  const cookieValue = cookieStore.get(INVITE_COOKIE_NAME)?.value
  const verified = cookieValue ? verifyInviteCookie(cookieValue, secret) : null

  if (verified) {
    try {
      const token = await getToken()
      const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'
      await fetch(`${apiUrl}/v1/auth/redeem-code`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({ code: verified.code }),
      })
    } catch {
      // Redemption failed — not fatal, user still gets in
    } finally {
      cookieStore.delete(INVITE_COOKIE_NAME)
    }
  }

  redirect('/events')
}
```
